📑 Cola de Comandos SQL

Clique em "Executar" em qualquer comando abaixo para rodar e visualizar instantaneamente:

  • -- Ver campanhas de usuários pendentes de moderação: SELECT id, telegram_id, titulo, link, cliques_totais, custo_total_pontos, status, data_criacao FROM campanhas_usuarios WHERE status = 'pendente' ORDER BY data_criacao DESC;
  • -- Funil real por campanha (Visitas, Cliques, Cadastros Reais, Tarefas e Indicações): WITH logs_por_campanha AS (SELECT COALESCE(utm_source, 'direct') AS utm_source, COALESCE(utm_medium, 'none') AS utm_medium, COALESCE(utm_campaign, 'none') AS utm_campaign, COUNT(*) AS total_visitas, SUM(CASE WHEN is_converted = true THEN 1 ELSE 0 END) AS cliques_botao_telegram FROM traffic_logs GROUP BY 1, 2, 3), usuarios_por_campanha AS (SELECT COALESCE(utm_source, 'direct') AS utm_source, COALESCE(utm_medium, 'none') AS utm_medium, COALESCE(utm_campaign, 'none') AS utm_campaign, COUNT(*) AS usuarios_cadastrados, SUM(CASE WHEN status_conta = 'ativo' THEN 1 ELSE 0 END) AS contas_ativas, SUM(CASE WHEN tarefas_feitas > 0 THEN 1 ELSE 0 END) AS usuarios_fizeram_tarefa, COALESCE(SUM(tarefas_feitas), 0) AS total_tarefas_concluidas, COALESCE(SUM(indicacoes), 0) AS total_indicacoes FROM usuarios GROUP BY 1, 2, 3) SELECT COALESCE(l.utm_source, u.utm_source) AS utm_source, COALESCE(l.utm_medium, u.utm_medium) AS utm_medium, COALESCE(l.utm_campaign, u.utm_campaign) AS utm_campaign, COALESCE(l.total_visitas, 0) AS total_visitas, COALESCE(l.cliques_botao_telegram, 0) AS cliques_convertidos_log, COALESCE(u.usuarios_cadastrados, 0) AS usuarios_reais_cadastrados, ROUND(COALESCE(u.usuarios_cadastrados, 0) * 100.0 / NULLIF(l.total_visitas, 0), 2) AS conv_visita_para_cadastro_pct, COALESCE(u.usuarios_fizeram_tarefa, 0) AS usuarios_engajados_tarefa, ROUND(COALESCE(u.usuarios_fizeram_tarefa, 0) * 100.0 / NULLIF(u.usuarios_cadastrados, 0), 2) AS conv_cadastro_para_tarefa_pct, COALESCE(u.total_tarefas_concluidas, 0) AS total_tarefas_feitas, COALESCE(u.total_indicacoes, 0) AS total_indicacoes FROM logs_por_campanha l FULL OUTER JOIN usuarios_por_campanha u ON l.utm_source = u.utm_source AND l.utm_medium = u.utm_medium AND l.utm_campaign = u.utm_campaign ORDER BY usuarios_reais_cadastrados DESC, total_visitas DESC;
  • -- Ver ações de um usuário específico: SELECT origem, COUNT(*) as total_acoes, SUM(pontos) as total_pontos FROM historico_ganhos WHERE telegram_id = '8513185549' GROUP BY origem ORDER BY total_acoes DESC;
  • -- Top 5 usuários por tarefas: SELECT nome, telegram_id, tarefas_feitas FROM usuarios ORDER BY tarefas_feitas DESC LIMIT 5;
  • -- Top 5 usuários por indicações: SELECT nome, telegram_id, indicacoes FROM usuarios ORDER BY indicacoes DESC LIMIT 5;
  • -- Diagnóstico de Bloqueios e Retenção (Novas colunas): SELECT COUNT(*) AS total_usuarios, SUM(CASE WHEN status_conta = 'bloqueado' THEN 1 ELSE 0 END) AS total_bloqueados, SUM(CASE WHEN status_conta = 'ativo' OR status_conta IS NULL THEN 1 ELSE 0 END) AS total_ativos, ROUND(AVG(tarefas_feitas), 2) AS media_tarefas_por_usuario, ROUND((SUM(CASE WHEN status_conta = 'bloqueado' THEN 1 ELSE 0 END)::decimal / NULLIF(COUNT(*), 0)) * 100, 2) AS taxa_bloqueio_percentual FROM usuarios;
